\begin{problem}
    Find $\frac{d}{dx}\left(x^3 + 2x^2 - 5x + 1\right)$.
\end{problem}

\begin{solution}
    Using the power rule:
    \begin{align*}
        \frac{d}{dx}\left(x^3 + 2x^2 - 5x + 1\right)
        &= 3x^2 + 4x - 5
    \end{align*}

    \answer{3x^2 + 4x - 5}
\end{solution}

\begin{problem}
    Evaluate $\int_0^1 x \cdot e^{x^2} \, \dd x$.
\end{problem}

\begin{solution}
    Let $u = x^2$, then $\dd u = 2x \, \dd x$:
    \begin{align*}
        \int_0^1 x \cdot e^{x^2} \, \dd x
        &= \frac{1}{2} \int_0^1 e^u \, \dd u \\
        &= \frac{1}{2} \left[ e^u \right]_0^1 \\
        &= \frac{1}{2}(e - 1)
    \end{align*}

    \answer{\frac{e - 1}{2}}
\end{solution}
